    <script>
        function calculateTax() {
            const country = document.getElementById('country').value;
            if (!country) { alert('Select a country'); return; }

            const gains = parseFloat(document.getElementById('gains').value) || 0;
            const staking = parseFloat(document.getElementById('staking').value) || 0;
            const freelance = parseFloat(document.getElementById('freelance').value) || 0;
            const losses = parseFloat(document.getElementById('losses').value) || 0;

            const totalIncome = gains + staking + freelance;
            const taxable = Math.max(0, totalIncome - losses);

            let tax = 0;
            if (country === 'us') {
                if (taxable <= 11600) tax = taxable * 0.10;
                else if (taxable <= 47150) tax = 1160 + (taxable - 11600) * 0.12;
                else tax = 5426 + (taxable - 47150) * 0.22;
                tax += taxable * 0.05; // state tax
                if (freelance > 0) tax += freelance * 0.153; // SE tax
            } else {
                if (taxable <= 55867) tax = taxable * 0.15;
                else if (taxable <= 111733) tax = 8380 + (taxable - 55867) * 0.205;
                else tax = 19630 + (taxable - 111733) * 0.26;
                tax += taxable * 0.08; // provincial
            }

            const quarterly = tax / 4;
            const monthly = tax / 12;

            document.getElementById('taxable').textContent = taxable.toFixed(2);
            document.getElementById('tax').textContent = tax.toFixed(2);
            document.getElementById('quarterly').textContent = quarterly.toFixed(2);
            document.getElementById('monthly').textContent = monthly.toFixed(2);
            document.getElementById('results').classList.add('show');
        }
    </script>
